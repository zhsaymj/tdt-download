## 0. 结论摘要

用户点名的三条路径——**大范围下载影像/地形**、**下载后合并与切片**、**选择本地大 tif**——现状是：

> **切片与合并的「读像素」环节已经全部是流式的，但「任务编排」环节仍在按瓦片总数累积对象。大范围任务仍会崩，崩因已从「整幅读入」转移到「按瓦片数线性累积」。**

项目一直在「逐个修好」：每次只修日志里报错的那一处，没有做过全仓普查。这是本报告的主要价值。

### 三条路径各自的结论

## 3. 确认安全的位置（防止后续误改）

这些地方**不要**为了「省内存」去改——它们已经是对的。

**本节按可达入口分组（第六轮重构）**：同一个模块可能既被 API import、又被 worker import，但「重活在哪跑」决定 OOM 打挂的是**整个服务**还是**单个任务**。故按 **A. API 进程 / B. worker 子进程 / C. 离线 CLI** 三组枚举；代码写法层面的通用做法（切片读、逐行缓冲、常量上限）跨入口生效，单列 D 组。每条都标出**判定依据的调用点**，便于回源核对。

> **为什么不用 import 关系判定**：实测 import 闭包 API 64 模块 / worker 59 / CLI 12 —— `backend/api/tasks.py:14` import `core.dem_tiling`、`:17` import `core.formats`，于是几乎每个 `backend/core/*` 都显示「API 可达」。这种口径没有区分度，本节一律按**实际执行位置**（谁在 `asyncio.to_thread` 或子进程里调它）判定。

三项**贯穿全脚本**的口径约定（它们正是前几轮数值反复的根因）：

1. **同名对象不同构造方式开销差 1.5~4 倍**，所以一律报「区间 + 采用值 + 构造口径」，不报单一「精确值」。`pathlib.Path` 尤其明显：`base / 逐段拼接` 446.80 B、`Path(完整字符串)` 605.80 B、相对短路径 438.80 B、`a/%d/%d/%d.png` 这种纯 ASCII 短路径 280.85 B —— 四者相差达 2.2 倍。
2. **测量期间不能持有被测对象之外的引用**。`asyncio.Task` 若在 `asyncio.run` 返回后才取样，事件循环本身会被计入，数值从 854 B 虚高到 1278 B —— 故一律在容器内部就地 `tracemalloc.start()`。

"""内存审计报告 §1 系数复现脚本（第六轮固化版）。

用法（在仓库根目录）：
    .venv/Scripts/python.exe -I _m.py            # 跑全部（不含 hillshade）

1. **每个系数都给出构造口径**。同名对象不同构造方式开销差 1.5~4 倍（`Path` 尤其明显），
   所以只报「区间 + 本报告采用值 + 为什么采用它」，不报单一「精确值」。
2. **测量期间不持有被测对象之外的引用**。事件循环、任务列表这类容器会把整条引用链
   一起计入（`asyncio.Task` 若在 `asyncio.run` 返回后仍持有任务，会把整个 loop 算进去，
   数值从 774 B 虚高到 1278 B），故一律在容器内部就地测量。
3. 只读：不修改任何业务代码；只在临时目录造数据，跑完即删。
"""
from __future__ import annotations

import asyncio
import concurrent.futures as cf
import gc
import gzip
import json
import os
import pathlib
import sys
import tracemalloc

ROOT = pathlib.Path(os.environ.get("AUDIT_ROOT") or pathlib.Path.cwd()).resolve()
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))

GiB = 2 ** 30
MiB = 2 ** 20


def _start():
    gc.collect()
    tracemalloc.start()
    tracemalloc.clear_traces()
    return tracemalloc.get_traced_memory()[0]


def _stop():
    cur, peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()
    return cur, peak


def _delta(fn):
    """在函数内部就地测量：返回 (存活增量, 峰值增量)，单位 B。"""
    base = _start()
    keep = fn()
    cur, peak = _stop()
    del keep
    gc.collect()
    return cur - base, peak - base


def _row(name, n, cur, peak=None, unit="个"):

# ------------------------------------------------------------- 2. asyncio.Task
def m_task(n=200_000):
    """在协程内部就地测量，避免把 loop 本身算进 Task 的开销。"""
    async def probe(drain):
        base = _start()
        tasks = [asyncio.create_task(asyncio.sleep(0)) for _ in range(n)]
        pending = tracemalloc.get_traced_memory()[0] - base
        if drain:
            await asyncio.gather(*tasks)
        done = tracemalloc.get_traced_memory()[0] - base
        peak = tracemalloc.get_traced_memory()[1] - base
        tracemalloc.stop()
        return pending, done, peak

    for drain in (False, True):
        p, d, k = asyncio.run(probe(drain))
        print("%-30s n=%-7d 创建后 %8.2f B | 结束持有 %8.2f B | 峰值 %8.2f B"
              % ("asyncio.Task(drain=%s)" % drain, n, p / n, d / n, k / n))

def m_hillshade(side=4000):
    """走真实签名 `hillshade_from_dem(dem_path, out_path)`：先落一张 int16 DEM GeoTIFF。"""
    import tempfile

    import numpy as np
    import rasterio
    from rasterio.transform import from_origin
    from backend.core.dem import hillshade_from_dem

    px = side * side
    tmp = pathlib.Path(tempfile.mkdtemp(prefix="lfma_hs_"))
    dem_path, out_path = tmp / "dem.tif", tmp / "hs.tif"
    arr = np.random.default_rng(0).integers(0, 3000, (side, side)).astype("int16")
    prof = {
        "driver": "GTiff", "width": side, "height": side, "count": 1,
        "dtype": "int16", "crs": "EPSG:3857",
        "transform": from_origin(12900000.0, 4900000.0, 30.0, 30.0),
    }
    with rasterio.open(dem_path, "w", **prof) as ds:
        ds.write(arr, 1)
    del arr
    gc.collect()

    def once(measure_peak):
        gc.collect()
        tracemalloc.start()
        tracemalloc.clear_traces()
        base = tracemalloc.get_traced_memory()[0]
        hillshade_from_dem(dem_path, out_path, build_overviews=False)
        cur, peak = tracemalloc.get_traced_memory()
        tracemalloc.stop()
        return (peak if measure_peak else cur) - base

    steady = once(False)
    peak = once(True)
    _row("hillshade 稳态", px, steady, unit="px")
    _row("hillshade 峰值", px, peak, unit="px")
    print("%-30s 单份 float64 = %.0f MiB，峰值/单份 = %.2f×"
          % ("hillshade", px * 8 / MiB, peak / (px * 8)))
    import shutil
    shutil.rmtree(tmp, ignore_errors=True)

# --------------------------------------------- 7. service_scan._Tree 路径均摊
class _Entry:
    """模拟 os.DirEntry：service_scan.py:138-153 只用到 path / name / suffix / is_dir / is_file。"""
    __slots__ = ("name", "path", "suffix", "_isdir")

    def __init__(self, path, isdir):
        self.path = str(path)
        self.name = path.name
        self.suffix = path.suffix
        self._isdir = isdir

    def is_dir(self, follow_symlinks=True):
        return self._isdir

    def is_file(self, follow_symlinks=True):
        return not self._isdir


class _Scan:
    def __init__(self, items):
        self._items = items

    def __enter__(self):
        return iter(self._items)

    def __exit__(self, *a):
        return False


def m_tree(tasks=20, xs=10, ys=20, cjk=True):
    """忠实复现 `backend/core/service_scan.py:116` 的 `_Tree`。

    只把 `os.scandir` 换成内存假目录项（不落盘），其余走真实实现。
    关键口径：`has_tile` / `tile_ext` 以**目录**为键（`:149-153`）—— 若错按文件建键，
    系数会涨到 1300 B 级，故本函数不做任何简化。
    """
    from backend.core import service_scan

    name = (lambda i: "任务%05d" % i) if cjk else (lambda i: "task%05d" % i)
    root = ROOT / "output"
    files, dirs = {}, {}
    dirs[str(root)] = []

    def add(parent, sub):
        pd = pathlib.Path(parent)
        sd = pd / sub
        dirs[str(pd)].append(_Entry(sd, True))
        dirs.setdefault(str(sd), [])
        files.setdefault(str(sd), [])
        return sd

    for t in range(tasks):
        td = add(root, name(t))
        zd = add(td, "tms")
        lv = add(zd, "13")
        for x in range(xs):
            xd = add(lv, str(x))
            for y in range(ys):
                files[str(xd)].append(_Entry(xd / ("%d.png" % y), False))

    n_dirs = len(dirs)
    n_files = sum(len(v) for v in files.values())
    n_paths = n_dirs + n_files

    orig = service_scan.os.scandir

    def fake(p):
        k = str(pathlib.Path(p))
        return _Scan(list(files.get(k, [])) + list(dirs.get(k, [])))

    service_scan.os.scandir = fake
    try:
        cur, peak = _delta(lambda: service_scan._Tree(root, 4))
    finally:
        service_scan.os.scandir = orig
    label = "service_scan._Tree(%s)" % ("CJK" if cjk else "ASCII")
    print("%-30s 目录 %d / 文件 %d / 路径 %d" % (label, n_dirs, n_files, n_paths))
    _row("  路径均摊", n_paths, cur, peak, unit="路径")


# ------------------------------------------------------------ 8. WKB 几何系数
def m_wkb():
    """单次分配会被 tracemalloc 的块粒度放大，故按 n 次取均。"""
    from shapely.geometry import LineString, Polygon

    def per_call(fn, n=1000):
        base = _start()
        keep = [fn() for _ in range(n)]
        cur, _ = _stop()
        del keep
        gc.collect()
        return (cur - base) / n

    for k in (4, 90):
        pts = [(i * 1e-4, i * 2e-4) for i in range(k)]
        got = per_call(lambda: LineString(pts).wkb)
        print("%-30s n=%-4d WKB = %7.1f B（9+16n = %d）" % ("LineString.wkb", k, got, 9 + 16 * k))
    ring = [(i * 1e-4, (i % 2) * 1e-4) for i in range(8)]
    got = per_call(lambda: Polygon(ring).wkb)
    print("%-30s n=%-4d WKB = %7.1f B（29+16n = %d）" % ("Polygon.wkb", 8, got, 29 + 16 * 8))
    pts = [(i * 1e-4, i * 2e-4) for i in range(90)]
    got = per_call(lambda: LineString(pts))
    print("%-30s %7.1f B（几何数据不在 Python 堆，getsizeof 无意义）"
          % ("LineString 对象", got))
